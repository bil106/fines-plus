import 'package:auto_route/auto_route.dart';
import 'package:core_localization/core_localization.dart';
import 'package:design_system/design_system.dart';
import 'package:flutter/material.dart';
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:tui_app/core/tui_bottom_nav_tab.dart';
import 'package:tui_app/core/widgets/top_divider.dart';
import 'package:tui_app/features/home/presentation/cubit/home_cubit.dart';
import 'package:tui_app/router/app_router.dart';

class TUIBottomNavigation extends StatelessWidget {
  const TUIBottomNavigation({super.key});

  final double iconSize = 24;

  @override
  Widget build(BuildContext context) {
    final l10n = AppLocalizations.of(context)!;

    return AutoTabsScaffold(
      routes: const [
        HomeRoute(),
        RechercheRoute(),
        AideRoute(),
        ReservationsRoute(),
        CompteRoute(),
      ],
      bottomNavigationBuilder: (_, tabsRouter) {
        return TopDivider(
          child: BottomNavigationBar(
            backgroundColor: Theme.of(context).scaffoldBackgroundColor,
            currentIndex: tabsRouter.activeIndex,
            onTap: (index) async {
              final selectedTab = TuiBottomNavTab.values[index];

              if (tabsRouter.activeIndex == index) {
                await _scrollHomeToTop(selectedTab, context);
              } else {
                tabsRouter.setActiveIndex(index);
                await _scrollHomeToTop(selectedTab, context);
              }
            },
            type: BottomNavigationBarType.fixed,
            selectedItemColor: TUIColors.energyBlue,
            unselectedItemColor: TUIColors.neutreGrey,
            items: [
              _buildNavItem(
                index: TuiBottomNavTab.home.index,
                tabsRouter: tabsRouter,
                svg: Assets.tab.smile,
                label: l10n.bottomNavHome,
              ),
              _buildNavItem(
                index: TuiBottomNavTab.search.index,
                tabsRouter: tabsRouter,
                svg: Assets.tab.search,
                label: l10n.searchTitle,
              ),
              _buildNavItem(
                index: TuiBottomNavTab.help.index,
                tabsRouter: tabsRouter,
                svg: Assets.tab.message,
                label: l10n.helpTitle,
              ),
              _buildNavItem(
                index: TuiBottomNavTab.reservations.index,
                tabsRouter: tabsRouter,
                svg: Assets.tab.luggage,
                label: l10n.reservationsTitle,
              ),
              _buildNavItem(
                index: TuiBottomNavTab.account.index,
                tabsRouter: tabsRouter,
                svg: Assets.tab.userCircled,
                label: l10n.accountTitle,
              ),
            ],
          ),
        );
      },
    );
  }

  Future<void> _scrollHomeToTop(
    TuiBottomNavTab selectedTab,
    BuildContext context,
  ) async {
    if (selectedTab == TuiBottomNavTab.home) {
      context.read<HomeCubit>().scrollToTop();
    }
  }

  BottomNavigationBarItem _buildNavItem({
    required int index,
    required TabsRouter tabsRouter,
    required SvgGenImage svg,
    required String label,
  }) {
    final isSelected = tabsRouter.activeIndex == index;

    return BottomNavigationBarItem(
      icon: svg.svg(
        colorFilter: ColorFilter.mode(
          isSelected ? TUIColors.energyBlue : TUIColors.neutreGrey,
          BlendMode.srcIn,
        ),
        fit: BoxFit.contain,
        width: iconSize,
        height: iconSize,
      ),
      label: label,
    );
  }
}
