import 'package:auto_route/auto_route.dart';
import 'package:flutter/material.dart';
import 'package:tui_app/features/auth/presentation/view/auth_screen.dart';
import 'package:tui_app/router/app_router.dart';

@RoutePage(name: 'AuthRoute')
class AuthScreenWrapper extends StatelessWidget {
  const AuthScreenWrapper({super.key});

  @override
  Widget build(BuildContext context) {
    return AuthScreen(
      onAuthenticated: () {
        context.replaceRoute(const TUIBottomNavigationWrapperRoute());
      },
    );
  }
}
