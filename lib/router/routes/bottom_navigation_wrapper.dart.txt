import 'package:auto_route/auto_route.dart';
import 'package:flutter/material.dart';
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:tui_app/features/home/presentation/cubit/home_cubit.dart';

@RoutePage(name: 'TUIBottomNavigationWrapperRoute')
class TUIBottomNavigationWrapper extends StatelessWidget {
  const TUIBottomNavigationWrapper({super.key});

  @override
  Widget build(BuildContext context) {
    return BlocProvider<HomeCubit>(
