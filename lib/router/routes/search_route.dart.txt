import 'package:auto_route/auto_route.dart';
import 'package:flutter/material.dart';
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:tui_app/core/cubits/location_cubit.dart';
import 'package:tui_app/features/search/domain/di/search_results_di.dart';
import 'package:tui_app/features/search/presentation/view/search_screen.dart';

@RoutePage(name: 'RechercheRoute')
class RechercheScreenWrapper extends StatefulWidget {
  const RechercheScreenWrapper({super.key});

  @override
  State<RechercheScreenWrapper> createState() => _RechercheScreenWrapperState();
}

class _RechercheScreenWrapperState extends State<RechercheScreenWrapper> {
  final di = SearchResultsDI(UniqueKey().toString());

  @override
  void initState() {
    di.init();
    super.initState();
  }

  @override
  void dispose() {
    di.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return BlocProvider(
      create: (_) => LocationCubit(),
      child: const SearchScreen(),
    );
  }
}
